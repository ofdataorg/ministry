External and anchor links pass through. */ -}} {{- $dest := .Destination -}} {{- if hasPrefix $dest "/" -}} {{- $dest = relURL (strings.TrimPrefix "/" $dest) -}} {{- end -}} {{ .Text | safeHTML }}
